```kotlin
package com.mucfy.injector

import android.content.Intent
import android.os.Bundle
import android.widget.Button
import android.widget.EditText
import android.widget.Toast
import androidx.appcompat.app.AppCompatActivity
import java.security.MessageDigest

class MainActivity : AppCompatActivity() {

    private val validKeyHash = "a3f5d7e9c1b2a4f6d8c0b1a5e9f7d3c2" // SHA-256 от "73D37C08"

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_main)

        val keyInput = findViewById<EditText>(R.id.key_input)
        val launchButton = findViewById<Button>(R.id.launch_button)

        launchButton.setOnClickListener {
            val inputKey = keyInput.text.toString().trim()
            if (validateKey(inputKey)) {
                // Сохраняем флаг успешной активации
                getSharedPreferences("license", MODE_PRIVATE)
                    .edit()
                    .putBoolean("activated", true)
                    .apply()

                // Запуск root-сервиса и игры
                startService(Intent(this, RootService::class.java))

                // Запуск Standoff 2
                val launchIntent = packageManager.getLaunchIntentForPackage("com.axlebolt.standoff2")
                if (launchIntent != null) {
                    startActivity(launchIntent)
                    finish() // Закрываем активити
                } else {
                    Toast.makeText(this, "Игра не установлена", Toast.LENGTH_LONG).show()
                }
            } else {
                Toast.makeText(this, "Неверный ключ", Toast.LENGTH_LONG).show()
            }
        }
    }

    private fun validateKey(input: String): Boolean {
        val digest = MessageDigest.getInstance("SHA-256")
        val hashBytes = digest.digest(input.toByteArray())
        val hexString = hashBytes.joinToString("") { "%02x".format(it) }
        return hexString == validKeyHash
    }
}
```

```kotlin
package com.mucfy.injector

import android.app.Notification
import android.app.NotificationChannel
import android.app.NotificationManager
import android.app.Service
import android.content.Context
import android.content.Intent
import android.os.Build
import android.os.IBinder
import android.os.Process
import java.io.DataOutputStream
import java.io.File
import java.io.FileOutputStream

class RootService : Service() {

    private val CHANNEL_ID = "InjectorChannel"
    private val NOTIFICATION_ID = 1

    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
        createNotificationChannel()
        startForeground(NOTIFICATION_ID, createNotification())

        // Запуск инжектора в отдельном потоке
        Thread {
            // Ждем запуска игры
            waitForGameProcess()
            // Выполняем инъекцию
            performInjection()
        }.start()

        return START_STICKY
    }

    private fun waitForGameProcess() {
        while (true) {
            val process = Runtime.getRuntime().exec("ps -A | grep com.axlebolt.standoff2")
            val output = process.inputStream.bufferedReader().readText()
            if (output.contains("com.axlebolt.standoff2")) {
                break
            }
            Thread.sleep(1000)
        }
    }

    private fun performInjection() {
        try {
            // 1. Копируем библиотеку в приватную директорию
            val libFile = File(filesDir, "libcheat.so")
            assets.open("libcheat.so").use { input ->
                FileOutputStream(libFile).use { output ->
                    input.copyTo(output)
                }
            }

            // 2. Получаем PID игры
            val pidProcess = Runtime.getRuntime().exec("pidof com.axlebolt.standoff2")
            val pid = pidProcess.inputStream.bufferedReader().readText().trim()

            // 3. Выполняем инъекцию через su
            val su = Runtime.getRuntime().exec("su")
            val outputStream = DataOutputStream(su.outputStream)

            // Команды для инъекции
            val commands = """
                setenforce 0
                cd ${filesDir.absolutePath}
                ./injector $pid libcheat.so
                setenforce 1
                exit
            """.trimIndent()

            outputStream.writeBytes(commands + "\n")
            outputStream.flush()
            outputStream.close()
            su.waitFor()

        } catch (e: Exception) {
            e.printStackTrace()
        }
    }

    private fun createNotificationChannel() {
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
            val channel = NotificationChannel(
                CHANNEL_ID,
                "Инжектор",
                NotificationManager.IMPORTANCE_LOW
            )
            val manager = getSystemService(NotificationManager::class.java)
            manager.createNotificationChannel(channel)
        }
    }

    private fun createNotification(): Notification {
        return Notification.Builder(this, CHANNEL_ID)
            .setContentTitle("Standoff2 Tools")
            .setContentText("Фоновая служба активна")
            .setSmallIcon(android.R.drawable.ic_dialog_info)
            .build()
    }

    override fun onBind(intent: Intent?): IBinder? = null
}
```

6. Нативный инжектор (injector.cpp):

```cpp
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <fcntl.h>
#include <sys/ptrace.h>
#include <sys/wait.h>
#include <sys/mman.h>
#include <dlfcn.h>
#include <android/log.h>

#define LOG_TAG "MucfyInjector"
#define LOGI(...) __android_log_print(ANDROID_LOG_INFO, LOG_TAG, __VA_ARGS__)
#define LOGE(...) __android_log_print(ANDROID_LOG_ERROR, LOG_TAG, __VA_ARGS__)

int main(int argc, char *argv[]) {
    if (argc != 3) {
        LOGI("Использование: %s <pid> <library.so>", argv[0]);
        return 1;
    }

    pid_t target_pid = atoi(argv[1]);
    const char *library_path = argv[2];

    LOGI("Начинаем инъекцию в PID: %d, библиотека: %s", target_pid, library_path);

    // 1. Присоединяемся к процессу через ptrace
    if (ptrace(PTRACE_ATTACH, target_pid, NULL, NULL) == -1) {
        LOGE("Не удалось присоединиться к процессу");
        return 1;
    }
    waitpid(target_pid, NULL, 0);

    // 2. Получаем адрес функции dlopen в целевом процессе
    void *local_handle = dlopen("libdl.so", RTLD_NOW);
    void *dlopen_addr = dlsym(local_handle, "dlopen");
    dlclose(local_handle);

    // 3. Подготавливаем строку с путем к библиотеке в памяти целевого процесса
    long mmap_addr = ptrace(PTRACE_PEEKTEXT, target_pid, 
                           ptrace(PTRACE_PEEKTEXT, target_pid, 
                           ptrace(PTRACE_PEEKTEXT, target_pid, 0, 0) + 0x10, 0) + 0x20, 0);

    // Упрощенный код: на практике требуется сложная логика аллокации памяти
    // через вызовы mmap и memcpy в контексте целевого процесса

    // 4. Вызываем dlopen в целевом процессе
    long regs_backup[32];
    ptrace(PTRACE_GETREGS, target_pid, NULL, regs_backup);

    // Устанавливаем регистры для вызова dlopen (архитектура arm64)
    struct user_pt_regs regs;
    memset(&regs, 0, sizeof(regs));
    regs.pc = (unsigned long)dlopen_addr;
    regs.regs[0] = mmap_addr; // Первый аргумент: путь к библиотеке
    regs.regs[1] = RTLD_NOW;  // Второй аргумент: флаги

    ptrace(PTRACE_SETREGS, target_pid, NULL, &regs);
    ptrace(PTRACE_CONT, target_pid, NULL, NULL);
    waitpid(target_pid, NULL, 0);

    // 5. Восстанавливаем регистры
    ptrace(PTRACE_SETREGS, target_pid, NULL, regs_backup);
    ptrace(PTRACE_DETACH, target_pid, NULL, NULL);

    LOGI("Инъекция завершена успешно");
    return 0;
}
```
